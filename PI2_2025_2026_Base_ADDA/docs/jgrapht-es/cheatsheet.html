📦 Imports Básicos

import org.jgrapht.*;
import org.jgrapht.graph.*;
import org.jgrapht.alg.shortestpath.DijkstraShortestPath;
import org.jgrapht.alg.spanning.KruskalMinimumSpanningTree;
import org.jgrapht.alg.color.GreedyColoring;
import org.jgrapht.traverse.*;
import org.jgrapht.alg.connectivity.ConnectivityInspector;

1️⃣ Crear Grafos

Grafo simple no dirigido

Graph<String, DefaultEdge> g = new SimpleGraph<>(DefaultEdge.class);

Grafo dirigido

Graph<String, DefaultEdge> dg = new SimpleDirectedGraph<>(DefaultEdge.class);

Grafo ponderado no dirigido

Graph<String, DefaultWeightedEdge> wg = new SimpleWeightedGraph<>(DefaultWeightedEdge.class);

Grafo dirigido y ponderado

Graph<String, DefaultWeightedEdge> dwg = new SimpleDirectedWeightedGraph<>(DefaultWeightedEdge.class);

2️⃣ Añadir Vértices y Aristas

Añadir vértice

g.addVertex("A");
g.addVertex("B");

Añadir arista (vértices deben existir)

g.addEdge("A", "B");

Añadir arista con vértices automáticos ⭐

// Crea los vértices si no existen
Graphs.addEdgeWithVertices(g, "C", "D");

Añadir arista ponderada con vértices ⭐

Graphs.addEdgeWithVertices(wg, "Madrid", "Barcelona", 621.0);

Establecer peso de arista

DefaultWeightedEdge e = wg.addEdge("A", "B");
wg.setEdgeWeight(e, 5.0);

3️⃣ Consultar Información

Obtener todos los vértices

Set<String> vertices = g.vertexSet();
System.out.println("Vértices: " + vertices);

Obtener todas las aristas

Set<DefaultEdge> aristas = g.edgeSet();
System.out.println("Número de aristas: " + aristas.size());

Verificar existencia de vértice/arista

boolean existe = g.containsVertex("A");
boolean conectados = g.containsEdge("A", "B");

Obtener grado de un vértice

int grado = g.degreeOf("A");  // Grafos no dirigidos

// Para grafos dirigidos:
int entrada = dg.inDegreeOf("A");   // Aristas entrantes
int salida = dg.outDegreeOf("A");   // Aristas salientes

4️⃣ Vecinos

Obtener vecinos de un vértice ⭐

List<String> vecinos = Graphs.neighborListOf(g, "A");
// o como Set:
Set<String> vecinosSet = Graphs.neighborSetOf(g, "A");

Predecesores y sucesores (grafos dirigidos)

// Vértices que apuntan a A
List<String> predecesores = Graphs.predecessorListOf(dg, "A");

// Vértices a los que A apunta
List<String> sucesores = Graphs.successorListOf(dg, "A");

5️⃣ Camino Mínimo (Dijkstra)

Encontrar camino más corto ⭐

DijkstraShortestPath<String, DefaultWeightedEdge> dijkstra = 
    new DijkstraShortestPath<>(wg);

GraphPath<String, DefaultWeightedEdge> camino = 
    dijkstra.getPath("origen", "destino");

if (camino != null) {
    System.out.println("Ruta: " + camino.getVertexList());
    System.out.println("Coste: " + camino.getWeight());
}

Solo obtener el peso del camino

double peso = dijkstra.getPathWeight("origen", "destino");

6️⃣ Árbol de Expansión Mínima (Kruskal)

Calcular MST ⭐

KruskalMinimumSpanningTree<String, DefaultWeightedEdge> kruskal = 
    new KruskalMinimumSpanningTree<>(wg);

SpanningTree<DefaultWeightedEdge> mst = kruskal.getSpanningTree();

System.out.println("Peso total: " + mst.getWeight());
System.out.println("Aristas: " + mst.getEdges());

7️⃣ Coloreado de Grafos

Colorear vértices (Greedy) ⭐

GreedyColoring<String, DefaultEdge> coloreo = 
    new GreedyColoring<>(g);

Coloring<String> resultado = coloreo.getColoring();

System.out.println("Colores necesarios: " + resultado.getNumberColors());

// Obtener color de cada vértice
Map<String, Integer> colores = resultado.getColors();
for (String v : g.vertexSet()) {
    System.out.println(v + " -> color " + colores.get(v));
}

8️⃣ Recorridos

Recorrido en anchura (BFS)

Iterator<String> bfs = new BreadthFirstIterator<>(g, "inicio");
while (bfs.hasNext()) {
    System.out.println(bfs.next());
}

Recorrido en profundidad (DFS)

Iterator<String> dfs = new DepthFirstIterator<>(g, "inicio");
while (dfs.hasNext()) {
    System.out.println(dfs.next());
}

9️⃣ Conectividad

Verificar si el grafo es conexo

ConnectivityInspector<String, DefaultEdge> inspector = 
    new ConnectivityInspector<>(g);

boolean esConexo = inspector.isConnected();
boolean hayRuta = inspector.pathExists("A", "Z");

Obtener componentes conexas

Set<Set<String>> componentes = inspector.connectedSets();
System.out.println("Componentes: " + componentes.size());

🔟 Operaciones con Aristas

Obtener extremos de una arista

DefaultEdge e = g.getEdge("A", "B");
String origen = g.getEdgeSource(e);
String destino = g.getEdgeTarget(e);

// O el vértice opuesto:
String otro = Graphs.getOppositeVertex(g, e, "A");  // -> "B"

Obtener aristas de un vértice

Set<DefaultEdge> aristasDeA = g.edgesOf("A");

// En grafos dirigidos:
Set<DefaultEdge> entrantes = dg.incomingEdgesOf("A");
Set<DefaultEdge> salientes = dg.outgoingEdgesOf("A");

📋 Ejemplo Completo

import org.jgrapht.*;
import org.jgrapht.graph.*;
import org.jgrapht.alg.shortestpath.DijkstraShortestPath;
import org.jgrapht.alg.spanning.KruskalMinimumSpanningTree;

public class EjemploCompleto {
    public static void main(String[] args) {
        // 1. Crear grafo ponderado
        Graph<String, DefaultWeightedEdge> g = 
            new SimpleWeightedGraph<>(DefaultWeightedEdge.class);
        
        // 2. Añadir aristas con vértices
        Graphs.addEdgeWithVertices(g, "A", "B", 4);
        Graphs.addEdgeWithVertices(g, "A", "C", 2);
        Graphs.addEdgeWithVertices(g, "B", "C", 1);
        Graphs.addEdgeWithVertices(g, "B", "D", 5);
        Graphs.addEdgeWithVertices(g, "C", "D", 3);
        
        // 3. Información básica
        System.out.println("Vértices: " + g.vertexSet());
        System.out.println("Aristas: " + g.edgeSet().size());
        
        // 4. Vecinos
        System.out.println("Vecinos de A: " + Graphs.neighborListOf(g, "A"));
        
        // 5. Camino mínimo
        DijkstraShortestPath<String, DefaultWeightedEdge> dijkstra = 
            new DijkstraShortestPath<>(g);
        GraphPath<String, DefaultWeightedEdge> camino = 
            dijkstra.getPath("A", "D");
        System.out.println("Camino A→D: " + camino.getVertexList());
        System.out.println("Coste: " + camino.getWeight());
        
        // 6. MST
        KruskalMinimumSpanningTree<String, DefaultWeightedEdge> kruskal = 
            new KruskalMinimumSpanningTree<>(g);
        System.out.println("Peso MST: " + kruskal.getSpanningTree().getWeight());
    }
}

⚡ Tips Rápidos

  • Usa Graphs.addEdgeWithVertices() siempre que puedas - es más seguro
  • Para grafos ponderados usa DefaultWeightedEdge y SimpleWeightedGraph
  • Dijkstra solo funciona con pesos no negativos
  • Kruskal solo funciona con grafos no dirigidos
  • El resultado de vertexSet() y edgeSet() son vistas del grafo
