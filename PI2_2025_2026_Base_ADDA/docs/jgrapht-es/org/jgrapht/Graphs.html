📖 Descripción

Graphs es una clase de utilidades que proporciona métodos estáticos para realizar operaciones comunes sobre grafos de forma más cómoda. Contiene funciones auxiliares muy útiles que simplifican tareas frecuentes.

💡 Tip: Todos los métodos de esta clase son static, por lo que se llaman directamente: Graphs.metodo(...)

🔧 Métodos Más Útiles

addEdgeWithVertices

static <V, E> E addEdgeWithVertices(Graph<V, E> g, V sourceVertex, V targetVertex)

Añade una arista entre dos vértices, creando los vértices automáticamente si no existen. Este es probablemente el método más útil de la clase. 

Graph<String, DefaultEdge> g = new SimpleGraph<>(DefaultEdge.class);

// Sin Graphs: necesitamos 3 líneas
g.addVertex("A");
g.addVertex("B");
g.addEdge("A", "B");

// Con Graphs: todo en 1 línea
Graphs.addEdgeWithVertices(g, "C", "D");  // Crea C, D y la arista
📝 Nota: Si los vértices ya existen, simplemente crea la arista. Nunca falla por vértices inexistentes.

addEdgeWithVertices (con peso)

static <V, E> E addEdgeWithVertices(Graph<V, E> g, V source, V target, double weight)

Versión para grafos ponderados: crea vértices, arista y asigna el peso en una sola operación.

Graph<String, DefaultWeightedEdge> g = 
    new SimpleWeightedGraph<>(DefaultWeightedEdge.class);

// Crea Madrid, Barcelona, la arista y le asigna peso 621
Graphs.addEdgeWithVertices(g, "Madrid", "Barcelona", 621.0);

neighborListOf

static <V, E> List<V> neighborListOf(Graph<V, E> g, V vertex)

Devuelve una lista con todos los vecinos de un vértice.

Graph<String, DefaultEdge> g = new SimpleGraph<>(DefaultEdge.class);
Graphs.addEdgeWithVertices(g, "Madrid", "Barcelona");
Graphs.addEdgeWithVertices(g, "Madrid", "Valencia");
Graphs.addEdgeWithVertices(g, "Madrid", "Sevilla");

List<String> vecinosMadrid = Graphs.neighborListOf(g, "Madrid");
// Resultado: [Barcelona, Valencia, Sevilla]

for (String vecino : vecinosMadrid) {
    System.out.println("Vecino de Madrid: " + vecino);
}

neighborSetOf

static <V, E> Set<V> neighborSetOf(Graph<V, E> g, V vertex)

Similar a neighborListOf pero devuelve un Set (sin duplicados).

Set<String> vecinos = Graphs.neighborSetOf(g, "Madrid");

predecessorListOf (grafos dirigidos)

static <V, E> List<V> predecessorListOf(Graph<V, E> g, V vertex)

En un grafo dirigido, devuelve los vértices que tienen aristas hacia el vértice dado.

Graph<String, DefaultEdge> dg = new SimpleDirectedGraph<>(DefaultEdge.class);
Graphs.addEdgeWithVertices(dg, "A", "C");  // A → C
Graphs.addEdgeWithVertices(dg, "B", "C");  // B → C

List<String> predecesores = Graphs.predecessorListOf(dg, "C");
// Resultado: [A, B]  (vértices que apuntan a C)

successorListOf (grafos dirigidos)

static <V, E> List<V> successorListOf(Graph<V, E> g, V vertex)

En un grafo dirigido, devuelve los vértices a los que apunta el vértice dado.

Graph<String, DefaultEdge> dg = new SimpleDirectedGraph<>(DefaultEdge.class);
Graphs.addEdgeWithVertices(dg, "A", "B");  // A → B
Graphs.addEdgeWithVertices(dg, "A", "C");  // A → C

List<String> sucesores = Graphs.successorListOf(dg, "A");
// Resultado: [B, C]  (vértices a los que A apunta)

getOppositeVertex

static <V, E> V getOppositeVertex(Graph<V, E> g, E e, V v)

Dada una arista y uno de sus vértices, devuelve el otro vértice de la arista.

DefaultEdge e = g.getEdge("Madrid", "Barcelona");
String otroExtremo = Graphs.getOppositeVertex(g, e, "Madrid");
// Resultado: "Barcelona"

vertexHasSuccessors / vertexHasPredecessors

static <V, E> boolean vertexHasSuccessors(Graph<V, E> g, V v)
static <V, E> boolean vertexHasPredecessors(Graph<V, E> g, V v)

Comprueba si un vértice tiene sucesores o predecesores (útil en grafos dirigidos).

if (Graphs.vertexHasSuccessors(dg, "A")) {
    System.out.println("A tiene aristas salientes");
}

if (!Graphs.vertexHasPredecessors(dg, "A")) {
    System.out.println("A es un vértice fuente (sin aristas entrantes)");
}

addAllVertices

static <V, E> boolean addAllVertices(Graph<V, E> destination, Collection<? extends V> vertices)

Añade múltiples vértices de una colección al grafo.

List<String> ciudades = Arrays.asList("Madrid", "Barcelona", "Valencia", "Sevilla");
Graphs.addAllVertices(g, ciudades);  // Añade las 4 ciudades

addGraph

static <V, E> boolean addGraph(Graph<? super V, ? super E> destination, Graph<V, E> source)

Copia todos los vértices y aristas de un grafo a otro.

Graph<String, DefaultEdge> original = ...;
Graph<String, DefaultEdge> copia = new SimpleGraph<>(DefaultEdge.class);
Graphs.addGraph(copia, original);  // copia = original

addGraphReversed

static <V, E> void addGraphReversed(Graph<? super V, ? super E> destination, Graph<V, E> source)

Copia un grafo invirtiendo la dirección de todas las aristas. Útil para grafos dirigidos.

Graph<String, DefaultEdge> original = new SimpleDirectedGraph<>(DefaultEdge.class);
Graphs.addEdgeWithVertices(original, "A", "B");  // A → B

Graph<String, DefaultEdge> invertido = new SimpleDirectedGraph<>(DefaultEdge.class);
Graphs.addGraphReversed(invertido, original);  // Ahora tiene B → A

addAllEdges

static <V, E> boolean addAllEdges(Graph<? super V, ? super E> dest, Graph<V, E> source, Collection<? extends E> edges)

Copia un conjunto de aristas de un grafo a otro.

Set<DefaultEdge> aristasSeleccionadas = ...;
Graphs.addAllEdges(destino, origen, aristasSeleccionadas);

addOutgoingEdges / addIncomingEdges

static <V, E> void addOutgoingEdges(Graph<V, E> g, V source, Iterable<V> targets)
static <V, E> void addIncomingEdges(Graph<V, E> g, V target, Iterable<V> sources)

Añade múltiples aristas salientes o entrantes a un vértice de una vez.

Graph<String, DefaultEdge> dg = new SimpleDirectedGraph<>(DefaultEdge.class);
List<String> destinos = Arrays.asList("B", "C", "D");

// Crea aristas A→B, A→C, A→D
Graphs.addOutgoingEdges(dg, "A", destinos);

// Crea aristas B→Z, C→Z, D→Z
List<String> origenes = Arrays.asList("B", "C", "D");
Graphs.addIncomingEdges(dg, "Z", origenes);

testIncidence

static <V, E> boolean testIncidence(Graph<V, E> g, E e, V v)

Comprueba si una arista es incidente a un vértice (es decir, si el vértice es uno de los extremos).

DefaultEdge e = g.getEdge("A", "B");
boolean esIncidente = Graphs.testIncidence(g, e, "A");  // true
boolean noIncidente = Graphs.testIncidence(g, e, "C");  // false

undirectedGraph

static <V, E> Graph<V, E> undirectedGraph(Graph<V, E> g)

Devuelve una vista no dirigida de un grafo dirigido. Las modificaciones en la vista afectan al original.

Graph<String, DefaultEdge> dirigido = new SimpleDirectedGraph<>(DefaultEdge.class);
// ... construir grafo dirigido ...

Graph<String, DefaultEdge> vistaNoDirigida = Graphs.undirectedGraph(dirigido);
// Ahora se puede recorrer ignorando la dirección de las aristas

removeVertexAndPreserveConnectivity

static <V, E> boolean removeVertexAndPreserveConnectivity(Graph<V, E> g, V vertex)

Elimina un vértice pero conecta sus predecesores con sus sucesores, preservando la conectividad del grafo.

// Grafo: A → B → C
Graph<String, DefaultEdge> dg = new SimpleDirectedGraph<>(DefaultEdge.class);
Graphs.addEdgeWithVertices(dg, "A", "B");
Graphs.addEdgeWithVertices(dg, "B", "C");

// Eliminar B conectando A directamente con C
Graphs.removeVertexAndPreserveConnectivity(dg, "B");
// Resultado: A → C

💡 Ejemplos Completos

Ejemplo 1: Construcción rápida de un grafo

import org.jgrapht.*;
import org.jgrapht.graph.*;

public class EjemploConstruccionRapida {
    public static void main(String[] args) {
        Graph<String, DefaultEdge> grafo = new SimpleGraph<>(DefaultEdge.class);
        
        // Forma rápida: una línea por arista
        Graphs.addEdgeWithVertices(grafo, "A", "B");
        Graphs.addEdgeWithVertices(grafo, "B", "C");
        Graphs.addEdgeWithVertices(grafo, "C", "D");
        Graphs.addEdgeWithVertices(grafo, "D", "A");
        Graphs.addEdgeWithVertices(grafo, "A", "C");
        
        System.out.println("Vértices: " + grafo.vertexSet());
        System.out.println("Aristas: " + grafo.edgeSet().size());
    }
}

Ejemplo 2: Obtener vecinos de un vértice

import org.jgrapht.*;
import org.jgrapht.graph.*;
import java.util.*;

public class EjemploVecinos {
    public static void main(String[] args) {
        Graph<String, DefaultEdge> red = new SimpleGraph<>(DefaultEdge.class);
        
        Graphs.addEdgeWithVertices(red, "Router", "PC1");
        Graphs.addEdgeWithVertices(red, "Router", "PC2");
        Graphs.addEdgeWithVertices(red, "Router", "PC3");
        Graphs.addEdgeWithVertices(red, "Router", "Servidor");
        
        // Obtener todos los dispositivos conectados al Router
        List<String> conectados = Graphs.neighborListOf(red, "Router");
        
        System.out.println("Dispositivos conectados al Router:");
        for (String dispositivo : conectados) {
            System.out.println("  - " + dispositivo);
        }
    }
}

Ejemplo 3: Grafo dirigido con predecesores y sucesores

import org.jgrapht.*;
import org.jgrapht.graph.*;
import java.util.*;

public class EjemploGrafoDirigido {
    public static void main(String[] args) {
        // Grafo de dependencias de tareas
        Graph<String, DefaultEdge> dependencias = 
            new SimpleDirectedGraph<>(DefaultEdge.class);
        
        // Tarea B depende de A: A → B
        Graphs.addEdgeWithVertices(dependencias, "Diseño", "Desarrollo");
        Graphs.addEdgeWithVertices(dependencias, "Desarrollo", "Testing");
        Graphs.addEdgeWithVertices(dependencias, "Diseño", "Documentación");
        Graphs.addEdgeWithVertices(dependencias, "Testing", "Despliegue");
        Graphs.addEdgeWithVertices(dependencias, "Documentación", "Despliegue");
        
        // ¿Qué tareas dependen de Diseño? (sucesores)
        List<String> dependientesDeDiseño = 
            Graphs.successorListOf(dependencias, "Diseño");
        System.out.println("Tareas que dependen de Diseño: " + dependientesDeDiseño);
        
        // ¿De qué tareas depende Despliegue? (predecesores)
        List<String> requisitosDespliegue = 
            Graphs.predecessorListOf(dependencias, "Despliegue");
        System.out.println("Requisitos de Despliegue: " + requisitosDespliegue);
    }
}
